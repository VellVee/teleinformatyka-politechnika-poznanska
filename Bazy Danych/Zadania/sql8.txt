SELECT nazwisko,
placa_pod, nazwa, 0.1*(avg(z.placa_pod))as podwyzka
FROM pracownicy p JOIN
zespoly z ON
p.id_zesp = z.id_zesp;

Select 0.1*Avg(placa_pod),id_zesp from pracownicy Group By id_zesp;

SELECT NAZWISKO, PLACA_POD, (SELECT avg(placa_pod) from pracownicy group by ID_ZESP) 
AS PODWYZKA FROM PRACOWNICY;

Select nazwisko,placa_pod, (Select 0.1*Avg(placa_pod) from pracownicy  
Where id_zesp=p.id_zesp Group By id_zesp) As podwyzka from pracownicy p;
