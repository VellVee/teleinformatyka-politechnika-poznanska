-- Wersja dla Oracle

DROP TABLE pacjenci CASCADE CONSTRAINTS;
DROP TABLE punkty_szczepien CASCADE CONSTRAINTS;
DROP TABLE rejestracje CASCADE CONSTRAINTS;
DROP TABLE szczepionki CASCADE CONSTRAINTS;
DROP TABLE szczepienia CASCADE CONSTRAINTS;


CREATE TABLE pacjenci( 
     pesel CHAR (11) constraint pacjenci_pk PRIMARY KEY, 
     imie VARCHAR2 (100) NOT NULL, 
     nazwisko VARCHAR2 (100) NOT NULL, 
     data_urodzenia DATE NOT NULL, 
     czy_choroby_wspolistniejace CHAR (1) NOT NULL constraint pacjenci_chk1 CHECK(czy_choroby_wspolistniejace IN ('T', 'N')));

CREATE TABLE punkty_szczepien ( 
     kod CHAR (10) constraint punkty_szczepien_pk PRIMARY KEY, 
     nazwa VARCHAR2 (50) NOT NULL, 
     adres VARCHAR2 (100) NOT NULL, 
     czy_dojazd_do_pacjenta CHAR(1) NOT NULL constraint punkty_szczepien_chk1 check(czy_dojazd_do_pacjenta in ('T','N')), 
     data_zgloszenia DATE NOT NULL);


CREATE TABLE rejestracje ( 
     pesel_pacjenta CHAR (11) NOT NULL 
		constraint rejestracje_fk1 REFERENCES pacjenci(pesel), 
     data DATE NOT NULL, 
     plan_termin_1_dawki DATE NOT NULL, 
     plan_termin_2_dawki DATE NOT NULL, 
     kod_punktu_szczepien CHAR(10) NOT NULL
		constraint rejestracje_fk2 REFERENCES punkty_szczepien(kod),
     constraint rejestracje_pk PRIMARY KEY (data, pesel_pacjenta));

CREATE TABLE szczepionki( 
     symbol VARCHAR2(10) constraint szczepionki_pk PRIMARY KEY, 
     nazwa VARCHAR2(100) NOT NULL, 
     producent VARCHAR2(100) NOT NULL, 
     czy_przechowywanie_ponizej_0 CHAR(1) NOT NULL 
		constraint szczepionki_chk1 CHECK(czy_przechowywanie_ponizej_0 IN ('T', 'N')));
	 
CREATE TABLE szczepienia( 
     data DATE NOT NULL, 
     symbol_szczepionki VARCHAR2(10) NOT NULL
		constraint szczepienia_fk1 REFERENCES szczepionki(symbol), 
     numer_dawki NUMBER (2) NOT NULL, 
     pesel_pacjenta CHAR(11) NOT NULL
		constraint szczepienia_fk2 REFERENCES pacjenci(pesel), 
     kod_punktu_szczepien CHAR(10) NOT NULL 
		constraint szczepienia_fk3 REFERENCES punkty_szczepien(kod),  
     constraint szczepienia_pk PRIMARY KEY(data, pesel_pacjenta));
